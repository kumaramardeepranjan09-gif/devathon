// Service budget measures commanded run time, not predicted failure life.
const double SERVICE_BUDGET_MINUTES = 200.0;

Preferences runtimeStore;
uint64_t runtimeMs1=0, runtimeMs2=0;
bool runtimePersistent=false;
unsigned long runtimeTick=0, runtimeSaved=0;
void accountRuntime() {
  unsigned long now=millis(), elapsed=now-runtimeTick; runtimeTick=now;
  if (driver1State && motor1State!="STOP" && speed1>0) runtimeMs1+=elapsed;
  if (driver2State && motor2State!="STOP" && speed2>0) runtimeMs2+=elapsed;
  if (runtimePersistent && now-runtimeSaved>=60000) {
    runtimePersistent=runtimeStore.putULong64("m1",runtimeMs1)==8 && runtimeStore.putULong64("m2",runtimeMs2)==8;
    runtimeSaved=now;
  }
}
void maintainWiFi() {
  // AutoReconnect owns retry scheduling. Never reconnect while STA is connecting.
  static bool wasConnected=false;
  static unsigned long lastNotice=0;
  bool connected=WiFi.status()==WL_CONNECTED;
  if(connected && !wasConnected){Serial.print("Dashboard IP: http://");Serial.println(WiFi.localIP());}
  if(!connected && wasConnected)Serial.println("Wi-Fi disconnected; automatic reconnection active, protection remains local.");
  if(!connected && millis()-lastNotice>=30000){lastNotice=millis();Serial.printf("Wi-Fi status %d; waiting for automatic connection\n",(int)WiFi.status());}
  wasConnected=connected;
}

// =============== MPU6050 REGISTERS =============
bool readReg(byte reg, byte &value) {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(reg);

  if (Wire.endTransmission(false) != 0) return false;

  if (Wire.requestFrom((uint8_t)MPU_ADDR,
                       (uint8_t)1,
                       (uint8_t)true) != 1) return false;

  value = Wire.read();
  return true;
}

bool readWord(byte reg, int16_t &value) {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(reg);

  if (Wire.endTransmission(false) != 0) return false;

  if (Wire.requestFrom((uint8_t)MPU_ADDR,
                       (uint8_t)2,
                       (uint8_t)true) != 2) return false;

  uint8_t hi = Wire.read();
  uint8_t lo = Wire.read();

  value = (int16_t)(((uint16_t)hi << 8) | lo);
  return true;
}

void initMPU() {
  byte who;

  if (!readReg(0x75, who)) {
    mpuOK = false;
    Serial.println("MPU6050 not detected");
    return;
  }

  Serial.printf("MPU WHO_AM_I: 0x%02X\n", who);

  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x6B);
  Wire.write(0x00);
  if (Wire.endTransmission() != 0) {
    mpuOK = false;
    return;
  }

  delay(100);

  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x1C);
  Wire.write(0x00);  // +/- 2g
  if (Wire.endTransmission() != 0) { mpuOK=false; return; }

  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x1B);
  Wire.write(0x00);  // +/- 250 degrees/s
  if (Wire.endTransmission() != 0) { mpuOK=false; return; }

  imuIdentity=who;
  mpuOK = (who == 0x68 || who == 0x70);
}

void readMPU() {
  if (!mpuOK) { static unsigned long retryAt=0; if(millis()-retryAt<2000)return;retryAt=millis();initMPU(); if (!mpuOK) return; }
  int16_t rax, ray, raz, rt, rgx, rgy, rgz;

  // One burst keeps all axes from the same register snapshot.
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B);
  if (Wire.endTransmission(false) != 0 || Wire.requestFrom((uint8_t)MPU_ADDR, (uint8_t)14, (uint8_t)true) != 14) {
    mpuOK=false; return;
  }
  int16_t words[7];
  for (int i=0;i<7;i++) { uint8_t hi=Wire.read(),lo=Wire.read(); words[i]=(int16_t)(((uint16_t)hi<<8)|lo); }
  rax=words[0]; ray=words[1]; raz=words[2]; rt=words[3]; rgx=words[4]; rgy=words[5]; rgz=words[6];

  mpuOK = true;

  ax = rax / 16384.0;
  ay = ray / 16384.0;
  az = raz / 16384.0;

  gx = rgx / 131.0;
  gy = rgy / 131.0;
  gz = rgz / 131.0;

  mpuTemp = imuIdentity==0x70 ? rt/333.87+21.0 : rt/340.0+36.53;

  float dx=ax-baseAx,dy=ay-baseAy,dz=az-baseAz;
  float magnitude=sqrtf(ax*ax+ay*ay+az*az);
  vibration=fabsf(magnitude-1.0f); // exclude static tilt from vibration magnitude
  filteredAx += .2f*(ax-filteredAx);filteredAy += .2f*(ay-filteredAy);filteredAz += .2f*(az-filteredAz);
  float baseNorm=sqrtf(baseAx*baseAx+baseAy*baseAy+baseAz*baseAz);
  float filteredNorm=sqrtf(filteredAx*filteredAx+filteredAy*filteredAy+filteredAz*filteredAz);
  if(imuCalibrated && baseNorm>.1f && filteredNorm>.1f){float cosine=(baseAx*filteredAx+baseAy*filteredAy+baseAz*filteredAz)/(baseNorm*filteredNorm);cosine=fmaxf(-1,fminf(1,cosine));tiltDeg=acosf(cosine)*180.0f/PI;}
  else tiltDeg=0;
  float rx=gx-baseGx,ry=gy-baseGy,rz=gz-baseGz;
  rotation=sqrtf(rx*rx+ry*ry+rz*rz);
  movementDetected=mpuOK && imuCalibrated && (vibration>MOTION_DETECT_G || rotation>MOTION_DETECT_DPS);
}

void recoverCalibration() {
  static int count=0;
  static double sx=0,sy=0,sz=0,sgx=0,sgy=0,sgz=0;
  static float lastAx=0,lastAy=0,lastAz=0;
  if(imuCalibrated)return;
  float norm=sqrtf(ax*ax+ay*ay+az*az);
  bool stable=mpuOK && !driver1State && !driver2State && norm>.75 && norm<1.25 && fabsf(gx)<15 && fabsf(gy)<15 && fabsf(gz)<15;
  if(count && (fabsf(ax-lastAx)+fabsf(ay-lastAy)+fabsf(az-lastAz)>.1))stable=false;
  lastAx=ax;lastAy=ay;lastAz=az;
  if(!stable){count=0;sx=sy=sz=sgx=sgy=sgz=0;return;}
  sx+=ax;sy+=ay;sz+=az;sgx+=gx;sgy+=gy;sgz+=gz;
  if(++count>=30){baseAx=sx/count;baseAy=sy/count;baseAz=sz/count;baseGx=sgx/count;baseGy=sgy/count;baseGz=sgz/count;filteredAx=baseAx;filteredAy=baseAy;filteredAz=baseAz;imuCalibrated=true;tiltDeg=0;Serial.println("IMU recovery baseline ready. Reset latch when safe; motors stay OFF.");}
}

// =============== TEMPERATURE UPDATES ===========
void updateTemperatures() {
  unsigned long now = millis();

  if (!tempPending && now - lastTempRequest >= 2500) {
    ds1.requestTemperatures();
    ds2.requestTemperatures();

    tempStarted = now;
    lastTempRequest = now;
    tempPending = true;
  }

  if (tempPending && now - tempStarted >= 800) {
    t1 = ds1.getTempCByIndex(0);
    t2 = ds2.getTempCByIndex(0);

    t1OK = t1 != DEVICE_DISCONNECTED_C &&
           isfinite(t1) && t1 >= -55 && t1 <= 125 && t1 != 85.0;

    t2OK = t2 != DEVICE_DISCONNECTED_C &&
           isfinite(t2) && t2 >= -55 && t2 <= 125 && t2 != 85.0;

    tempPending = false;
    lastGoodTemp = now;
  }
}

// =============== SAFETY ========================
void criticalShutdown(String reason) {
  if (!criticalLock) {
    criticalLock = true;
    criticalReason = reason;
    logEvent("CRITICAL: " + reason);
  }

  stopAll();
  driver1(false);
  driver2(false);
  digitalWrite(BUZZER_PIN, HIGH);
}

void evaluateSafety() {
  if (millis() > 5000 && (!t1OK || !t2OK || !mpuOK || !imuCalibrated || millis()-lastGoodTemp > 5000)) {
    criticalShutdown((!mpuOK || !imuCalibrated) ? "Shared IMU fault or baseline missing: restart with sensor still" : !t1OK ? "Motor 1 temperature sensor fault" : !t2OK ? "Motor 2 temperature sensor fault" : "Temperature samples stale");
  }
  bool tempCritical =
    (t1OK && t1 >= TEMP_CRITICAL_C) ||
    (t2OK && t2 >= TEMP_CRITICAL_C);

  bool tempWarning =
    (t1OK && t1 >= TEMP_WARNING_C) ||
    (t2OK && t2 >= TEMP_WARNING_C);

  bool motionWarning =
    imuCalibrated && tiltDeg >= TILT_WARNING_DEG;

  bool motionCritical =
    imuCalibrated && tiltDeg >= TILT_CRITICAL_DEG;

  if (motionWarning) { if (warningMotionSamples < 3) warningMotionSamples++; }
  else warningMotionSamples = 0;

  if (motionCritical) { if (criticalMotionSamples < 4) criticalMotionSamples++; }
  else criticalMotionSamples = 0;

  if (tempCritical) {
    criticalShutdown(t1OK && t1 >= TEMP_CRITICAL_C ? "Motor 1 temperature critical" : "Motor 2 temperature critical");
  } else if (criticalMotionSamples >= 4) {
    criticalShutdown("Shared IMU tilt reached critical limit");
  }

  if (criticalLock) {
    statusText = "CRITICAL";
    warningText = criticalReason;
  } else if (tempWarning || warningMotionSamples >= 3) {
    statusText = "WARNING";
    warningText = tempWarning
      ? (t1OK && t1 >= TEMP_WARNING_C ? "Motor 1 temperature warning" : "Motor 2 temperature warning")
      : "Shared IMU tilt warning";
  } else if (!t1OK || !t2OK || !mpuOK) {
    statusText = "WARNING";
    warningText = "Check sensor connection";
  } else {
    statusText = "NORMAL";
    warningText = "System operating normally";
  }

  if (!criticalLock) digitalWrite(BUZZER_PIN, LOW);
}

// =============== WEB COMMANDS ==================
void handleCommand() {
  accountRuntime();
  if (!server.hasArg("c")) {
    server.send(400, "text/plain", "Missing command");
    return;
  }

  if (server.header("Origin").length() && server.header("Origin") != "http://" + server.hostHeader()) {
    server.send(403,"text/plain","Cross-origin command blocked"); return;
  }
  String c = server.arg("c");
  c.toUpperCase();

  if (criticalLock && c != "RESET" && c != "ALLSTOP" && c != "M1S" && c != "M2S") {
    server.send(423, "text/plain",
                "CRITICAL LOCK: resolve fault and reset");
    return;
  }

  bool starts = c == "M1F" || c == "M1R" || c == "M2F" || c == "M2R";
  if (starts && (!t1OK || !t2OK || !mpuOK || !imuCalibrated || t1 >= TEMP_WARNING_C || t2 >= TEMP_WARNING_C || tiltDeg >= TILT_WARNING_DEG)) {
    server.send(409,"text/plain","Start blocked: sensors must be healthy and below warning limits"); return;
  }
  if (c == "M1F") m1Forward();
  else if (c == "M1R") m1Reverse();
  else if (c == "M1S") stopM1();
  else if (c == "M2F") m2Forward();
  else if (c == "M2R") m2Reverse();
  else if (c == "M2S") stopM2();
  else if (c == "ALLSTOP") { stopAll(); driver1(false); driver2(false); }
  else if (c == "SPEED1" || c == "SPEED2") {
    if (!server.hasArg("v")) {
      server.send(400, "text/plain", "Missing speed");
      return;
    }

    String raw = server.arg("v");
    for (unsigned int i=0;i<raw.length();i++) if (!isDigit(raw[i])) { server.send(400,"text/plain","Invalid PWM"); return; }
    int v = raw.toInt();
    if (!raw.length() || raw.length()>3 || v<0 || v>255) { server.send(400,"text/plain","PWM must be 0..255"); return; }

    if (c == "SPEED1") {
      speed1 = v;
      if (motor1State == "FORWARD") m1Forward();
      else if (motor1State == "REVERSE") m1Reverse();
    } else {
      speed2 = v;
      if (motor2State == "FORWARD") m2Forward();
      else if (motor2State == "REVERSE") m2Reverse();
    }
  } else if (c == "RESET") {
    bool tempSafe =
      (t1OK && t1 < TEMP_RESET_C) &&
      (t2OK && t2 < TEMP_RESET_C);

    bool motionSafe =
      imuCalibrated && tiltDeg < TILT_WARNING_DEG;

    if (!tempSafe || !motionSafe || !mpuOK) {
      server.send(409, "text/plain",
                  "Reset blocked: check sensors/conditions");
      return;
    }

    criticalLock = false;
    criticalReason = "";
    criticalMotionSamples = 0;
    warningMotionSamples = 0;
    digitalWrite(BUZZER_PIN, LOW);
    logEvent("Safety reset");
  } else {
    server.send(400, "text/plain", "Unknown command");
    return;
  }

  logEvent("Command: " + c);
  server.send(200, "text/plain", "OK");
}

// Service protection between network blocks; never recursively handle HTTP.
void serviceSensors() {
  accountRuntime();
  updateTemperatures();
  unsigned long now=millis();
  if(now-lastMPU>=100){lastMPU=now;readMPU();recoverCalibration();sampleSequence++;evaluateSafety();}
}

// NetworkClient may accept fewer bytes than requested. Advance only by bytes
// actually written, with bounded retries and cooperative protection updates.
bool writeAll(NetworkClient &client, const uint8_t *bytes, size_t length) {
  size_t offset=0;
  unsigned long lastProgress=millis();
  while(offset<length && client.connected()) {
    serviceSensors();
    size_t amount=length-offset; if(amount>1024)amount=1024;
    size_t written=client.write(bytes+offset,amount);
    if(written>amount)return false;
    if(written){offset+=written;lastProgress=millis();}
    else if(millis()-lastProgress>5000)break;
    delay(1);
  }
  return offset==length;
}
void serveAsset(const uint8_t *bytes, size_t length, const char *mime, bool cache) {
  NetworkClient client=server.client();
  client.setTimeout(1000);
  String header="HTTP/1.1 200 OK\r\nContent-Type: "+String(mime)+"\r\nContent-Encoding: gzip\r\nContent-Length: "+String(length)+"\r\nCache-Control: "+String(cache?"public, max-age=31536000, immutable":"no-store")+"\r\nConnection: close\r\n\r\n";
  Serial.printf("HTTP transfer: %s (%u bytes), free heap %u\n",mime,(unsigned)length,ESP.getFreeHeap());
  bool ok=writeAll(client,(const uint8_t*)header.c_str(),header.length()) && writeAll(client,bytes,length);
  Serial.println(ok?"HTTP transfer complete":"HTTP transfer interrupted; reload page and check power/Wi-Fi");
  client.stop();
}


// =============== SETUP =========================
void setup() {
  Serial.begin(115200);
  Serial.printf("Boot reset reason: %d; free heap: %u\n",(int)esp_reset_reason(),ESP.getFreeHeap());
  delay(700);

  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(ENB, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  bool pwm1=ledcAttach(ENA,1000,8), pwm2=ledcAttach(ENB,1000,8);
  if (!pwm1 || !pwm2) { Serial.println("PWM initialization failed; motor outputs locked OFF"); while(true) { digitalWrite(IN1,LOW);digitalWrite(IN2,LOW);digitalWrite(IN3,LOW);digitalWrite(IN4,LOW);delay(1000); } }
  stopAll();

  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);

  ds1.begin();
  ds2.begin();
  ds1.setWaitForConversion(false);
  ds2.setWaitForConversion(false);

  Wire.begin(MPU_SDA, MPU_SCL);
  Wire.setClock(100000);
  initMPU();
  if(mpuOK){
    Serial.println("Keep shared IMU still: baseline calibration, motors OFF.");
    double sx=0,sy=0,sz=0,sgx=0,sgy=0,sgz=0;int samples=0;
    for(int i=0;i<100;i++){readMPU();if(!mpuOK)break;sx+=ax;sy+=ay;sz+=az;sgx+=gx;sgy+=gy;sgz+=gz;samples++;delay(10);}
    float baselineNorm=sqrt(sx*sx+sy*sy+sz*sz)/(samples?samples:1);
    if(samples==100 && baselineNorm>.75 && baselineNorm<1.25){baseAx=sx/samples;baseAy=sy/samples;baseAz=sz/samples;baseGx=sgx/samples;baseGy=sgy/samples;baseGz=sgz/samples;filteredAx=baseAx;filteredAy=baseAy;filteredAz=baseAz;imuCalibrated=true;readMPU();Serial.println("IMU baseline stored for this boot.");}
    else {mpuOK=false;criticalShutdown("IMU baseline failed; restart with sensor connected");}
  }

  runtimePersistent=runtimeStore.begin("motor-runtime",false);
  if (runtimePersistent) { runtimeMs1=runtimeStore.getULong64("m1",0); runtimeMs2=runtimeStore.getULong64("m2",0); }
  runtimeTick=runtimeSaved=millis();
  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  WiFi.setAutoReconnect(true);
  WiFi.onEvent([](WiFiEvent_t event, WiFiEventInfo_t info){
    if(event==ARDUINO_EVENT_WIFI_STA_DISCONNECTED)Serial.printf("Wi-Fi disconnect reason: %u\n",info.wifi_sta_disconnected.reason);
  });
  WiFi.begin(WIFI_NAME,WIFI_PASSWORD);
  Serial.println("Connecting to configured Wi-Fi. Dashboard IP prints when connected.");

  server.on("/", HTTP_GET, []() {
    serveAsset(PAGE_GZ,sizeof(PAGE_GZ),"text/html; charset=utf-8",false);
  });

  server.on("/three-part", HTTP_GET, [](){
    String raw=server.arg("i");
    if(!raw.length()){server.send(400,"text/plain","Missing index");return;}
    for(unsigned i=0;i<raw.length();i++)if(!isDigit(raw[i])){server.send(400,"text/plain","Invalid index");return;}
    int index=raw.toInt();
    if(index<0 || index>=THREE_PART_COUNT){server.send(404,"text/plain","Unknown part");return;}
    serveAsset(THREE_PARTS[index],THREE_PART_SIZES[index],"application/octet-stream",false);
  });
  server.on("/api", HTTP_GET, handleAPI);
  const char* headers[] = {"Origin"};
  server.collectHeaders(headers, 1);
  server.on("/cmd", HTTP_POST, handleCommand);
  server.begin();

  Serial.println("Web server started");
  Serial.println("Waiting for phone connection...");

  ds1.requestTemperatures();
  ds2.requestTemperatures();
  tempStarted = millis();
  lastTempRequest = millis();
  tempPending = true;
}

// =============== LOOP ==========================
void loop() {
  serviceSensors();
  maintainWiFi();
  server.handleClient();
}
